'use client'
import React, { useState } from 'react';
import { X, CheckCircle2, ArrowRight, ArrowLeft, ShieldCheck, HeartHandshake, Sparkles, Building2, UserCheck } from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';
import { useRouter } from 'next/navigation';

export const Modals: React.FC = () => {
  const { activeModal, closeModal, push } = useRouter();

  // Assessment Modal state
  const [step, setStep] = useState(1);
  const [seekingFor, setSeekingFor] = useState<'myself' | 'family' | 'patient'>('myself');
  const [careType, setCareType] = useState<string>('Neurological & Stroke Recovery');
  const [timeline, setTimeline] = useState<string>('Immediate (within 48 hours)');
  const [contactInfo, setContactInfo] = useState({ name: '', phone: '', email: '' });
  const [submitted, setSubmitted] = useState(false);

  // Reset modal state on close
  const handleClose = () => {
    closeModal();
    setTimeout(() => {
      setStep(1);
      setSubmitted(false);
      setContactInfo({ name: '', phone: '', email: '' });
    }, 300);
  };

  const handleAssessmentSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setSubmitted(true);
  };

  if (!activeModal) return null;

  return (
    <AnimatePresence>
      <div className="fixed inset-0 z-50 flex items-center justify-center p-4 sm:p-6 overflow-y-auto">
        {/* Backdrop */}
        <motion.div
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          onClick={handleClose}
          className="fixed inset-0 bg-slate-950/70 backdrop-blur-sm"
        />

        {/* Modal Container */}
        <motion.div
          initial={{ opacity: 0, scale: 0.95, y: 10 }}
          animate={{ opacity: 1, scale: 1, y: 0 }}
          exit={{ opacity: 0, scale: 0.95, y: 10 }}
          className="relative w-full max-w-xl bg-white dark:bg-slate-900 rounded-3xl shadow-2xl border border-slate-200 dark:border-slate-800 p-6 sm:p-8 z-10 overflow-hidden my-8"
        >
          <button
            onClick={handleClose}
            className="absolute top-5 right-5 p-2 rounded-full text-slate-400 hover:text-slate-600 dark:hover:text-white bg-slate-100 dark:bg-slate-800 transition-colors"
          >
            <X className="w-5 h-5" />
          </button>

          {activeModal === 'assessment' && (
            <div>
              {!submitted ? (
                <div>
                  <div className="flex items-center gap-2 text-teal-600 dark:text-teal-400 text-xs font-semibold uppercase tracking-wider mb-2">
                    <Sparkles className="w-4 h-4" />
                    <span>Step {step} of 3 — Confidential Recovery Quiz</span>
                  </div>

                  <h3 className="font-heading font-bold text-2xl text-slate-900 dark:text-white">
                    {step === 1 && 'Who is this rehabilitation support for?'}
                    {step === 2 && 'What primary care area is needed?'}
                    {step === 3 && 'Where should your coordinator send your plan?'}
                  </h3>
                  <p className="text-sm text-slate-600 dark:text-slate-400 mt-1 mb-6">
                    {step === 1 && 'We match you with specialists suited for your specific family or personal situation.'}
                    {step === 2 && 'Select the main clinical focus so we can route to accredited facilities.'}
                    {step === 3 && 'Your details are strictly confidential and only shared with your assigned lead coordinator.'}
                  </p>

                  {/* Step 1 */}
                  {step === 1 && (
                    <div className="space-y-3">
                      {[
                        { id: 'myself', title: 'For Myself', desc: 'Seeking direct personal therapy, detox, or rehabilitation care.' },
                        { id: 'family', title: 'For a Family Member / Friend', desc: 'Helping a parent, sibling, spouse, or friend navigate care options.' },
                        { id: 'patient', title: 'Clinical Referral / Healthcare Provider', desc: 'Referral on behalf of a discharge hospital or medical team.' },
                      ].map((option) => (
                        <button
                          key={option.id}
                          type="button"
                          onClick={() => setSeekingFor(option.id as any)}
                          className={`w-full text-left p-4 rounded-2xl border transition-all ${
                            seekingFor === option.id
                              ? 'border-teal-500 bg-teal-50/50 dark:bg-teal-950/40 ring-2 ring-teal-500/20'
                              : 'border-slate-200 dark:border-slate-800 hover:bg-slate-50 dark:hover:bg-slate-800/50'
                          }`}
                        >
                          <div className="flex items-center justify-between">
                            <span className="font-heading font-semibold text-slate-900 dark:text-white text-base">
                              {option.title}
                            </span>
                            {seekingFor === option.id && <CheckCircle2 className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
                          </div>
                          <p className="text-xs text-slate-500 dark:text-slate-400 mt-1">{option.desc}</p>
                        </button>
                      ))}

                      <div className="pt-4 flex justify-end">
                        <button
                          onClick={() => setStep(2)}
                          className="px-6 py-3 rounded-xl bg-teal-600 hover:bg-teal-500 text-white font-semibold text-sm flex items-center gap-2 shadow-md"
                        >
                          <span>Next Step</span>
                          <ArrowRight className="w-4 h-4" />
                        </button>
                      </div>
                    </div>
                  )}

                  {/* Step 2 */}
                  {step === 2 && (
                    <div className="space-y-4">
                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2.5">
                        {[
                          'Neurological & Stroke Recovery',
                          'Substance Dependency & Detox',
                          'Spinal Cord & Physical Trauma',
                          'Post-Surgical Orthopedic Rehab',
                          'Burnout & Psychological Wellness',
                          'Geriatric & Cognitive Care',
                        ].map((type) => (
                          <button
                            key={type}
                            type="button"
                            onClick={() => setCareType(type)}
                            className={`p-3 rounded-xl border text-left text-xs font-semibold transition-all ${
                              careType === type
                                ? 'border-teal-500 bg-teal-50 dark:bg-teal-950/40 text-teal-800 dark:text-teal-300'
                                : 'border-slate-200 dark:border-slate-800 text-slate-700 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800'
                            }`}
                          >
                            {type}
                          </button>
                        ))}
                      </div>

                      <div className="pt-2">
                        <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1.5">
                          Desired Placement Timeline
                        </label>
                        <select
                          value={timeline}
                          onChange={(e) => setTimeline(e.target.value)}
                          className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs font-medium text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                        >
                          <option>Immediate (within 48 hours)</option>
                          <option>Within 1 - 2 weeks</option>
                          <option>Planning ahead (this month)</option>
                          <option>Just seeking preliminary information</option>
                        </select>
                      </div>

                      <div className="pt-4 flex items-center justify-between">
                        <button
                          type="button"
                          onClick={() => setStep(1)}
                          className="px-4 py-2.5 rounded-xl text-xs font-semibold text-slate-600 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 flex items-center gap-1.5"
                        >
                          <ArrowLeft className="w-4 h-4" /> Back
                        </button>
                        <button
                          type="button"
                          onClick={() => setStep(3)}
                          className="px-6 py-3 rounded-xl bg-teal-600 hover:bg-teal-500 text-white font-semibold text-sm flex items-center gap-2 shadow-md"
                        >
                          <span>Final Step</span>
                          <ArrowRight className="w-4 h-4" />
                        </button>
                      </div>
                    </div>
                  )}

                  {/* Step 3 */}
                  {step === 3 && (
                    <form onSubmit={handleAssessmentSubmit} className="space-y-4">
                      <div>
                        <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                          Full Name *
                        </label>
                        <input
                          type="text"
                          required
                          placeholder="e.g. Dr. Jane Smith or Chidi Okafor"
                          value={contactInfo.name}
                          onChange={(e) => setContactInfo({ ...contactInfo, name: e.target.value })}
                          className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white placeholder-slate-400 focus:outline-none focus:border-teal-500"
                        />
                      </div>

                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                        <div>
                          <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                            Phone / WhatsApp Number *
                          </label>
                          <input
                            type="tel"
                            required
                            placeholder="+234 800 000 0000"
                            value={contactInfo.phone}
                            onChange={(e) => setContactInfo({ ...contactInfo, phone: e.target.value })}
                            className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white placeholder-slate-400 focus:outline-none focus:border-teal-500"
                          />
                        </div>

                        <div>
                          <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                            Email Address *
                          </label>
                          <input
                            type="email"
                            required
                            placeholder="name@example.com"
                            value={contactInfo.email}
                            onChange={(e) => setContactInfo({ ...contactInfo, email: e.target.value })}
                            className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white placeholder-slate-400 focus:outline-none focus:border-teal-500"
                          />
                        </div>
                      </div>

                      <div className="p-3 rounded-xl bg-slate-100 dark:bg-slate-800/70 text-[11px] text-slate-500 dark:text-slate-400 flex items-start gap-2">
                        <ShieldCheck className="w-4 h-4 text-teal-500 shrink-0 mt-0.5" />
                        <span>
                          Your information is encrypted. A Care Coordinator will contact you confidentially within 2 hours.
                        </span>
                      </div>

                      <div className="pt-3 flex items-center justify-between">
                        <button
                          type="button"
                          onClick={() => setStep(2)}
                          className="px-4 py-2.5 rounded-xl text-xs font-semibold text-slate-600 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 flex items-center gap-1.5"
                        >
                          <ArrowLeft className="w-4 h-4" /> Back
                        </button>
                        <button
                          type="submit"
                          className="px-6 py-3 rounded-xl bg-teal-600 hover:bg-teal-500 text-white font-semibold text-sm shadow-lg shadow-teal-600/25 flex items-center gap-2"
                        >
                          <span>Request Assessment Plan</span>
                          <CheckCircle2 className="w-4 h-4" />
                        </button>
                      </div>
                    </form>
                  )}
                </div>
              ) : (
                /* Confirmation Screen */
                <div className="text-center py-6 space-y-4">
                  <div className="w-16 h-16 rounded-full bg-teal-100 dark:bg-teal-950/80 text-teal-600 dark:text-teal-400 flex items-center justify-center mx-auto ring-8 ring-teal-50 dark:ring-teal-900/30">
                    <CheckCircle2 className="w-8 h-8" />
                  </div>

                  <h3 className="font-heading font-bold text-2xl text-slate-900 dark:text-white">
                    Assessment Request Received!
                  </h3>

                  <p className="text-sm text-slate-600 dark:text-slate-300 max-w-md mx-auto leading-relaxed">
                    Thank you, <strong className="text-slate-900 dark:text-white">{contactInfo.name || 'Friend'}</strong>. Senior Care Coordinator <strong className="text-teal-600 dark:text-teal-400">Dr. Amara Okafor</strong> has been assigned to your confidential inquiry ({careType}).
                  </p>

                  <div className="p-4 rounded-2xl bg-slate-50 dark:bg-slate-800/50 border border-slate-200 dark:border-slate-700/60 text-left text-xs space-y-2">
                    <div className="flex justify-between font-medium">
                      <span className="text-slate-500">Care Focus:</span>
                      <span className="text-slate-900 dark:text-white font-semibold">{careType}</span>
                    </div>
                    <div className="flex justify-between font-medium">
                      <span className="text-slate-500">Timeline:</span>
                      <span className="text-slate-900 dark:text-white font-semibold">{timeline}</span>
                    </div>
                    <div className="flex justify-between font-medium">
                      <span className="text-slate-500">Status:</span>
                      <span className="text-emerald-600 dark:text-emerald-400 font-semibold flex items-center gap-1">
                        <span className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span> Routing to Coordinator
                      </span>
                    </div>
                  </div>

                  <div className="pt-4 flex justify-center gap-3">
                    <button
                      onClick={handleClose}
                      className="px-6 py-3 rounded-xl bg-slate-900 dark:bg-white text-white dark:text-slate-900 font-semibold text-sm shadow-md"
                    >
                      Return to Website
                    </button>
                  </div>
                </div>
              )}
            </div>
          )}

          {activeModal === 'partner' && (
            <div className="space-y-4">
              <div className="flex items-center gap-2 text-teal-600 dark:text-teal-400 text-xs font-semibold uppercase tracking-wider">
                <Building2 className="w-4 h-4" />
                <span>Rehabilitation Centre Partnership</span>
              </div>
              <h3 className="font-heading font-bold text-2xl text-slate-900 dark:text-white">
                Apply to Join the RehabConnect Verified Network
              </h3>
              <p className="text-sm text-slate-600 dark:text-slate-400">
                Receive pre-assessed patient referrals, expand your facility reach, and collaborate with certified Care Coordinators.
              </p>

              <form
                onSubmit={(e) => {
                  e.preventDefault();
                  alert('Thank you! Our partner onboarding team will contact your facility director within 24 hours.');
                  handleClose();
                }}
                className="space-y-3 pt-2"
              >
                <div>
                  <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                    Facility Name *
                  </label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. Gracefield Recovery Sanctuary"
                    className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                  />
                </div>

                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                      City / Location *
                    </label>
                    <input
                      type="text"
                      required
                      placeholder="e.g. Lagos, Abuja..."
                      className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                      Bed Capacity *
                    </label>
                    <input
                      type="text"
                      required
                      placeholder="e.g. 30 Beds"
                      className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                    />
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                    Medical Director Email *
                  </label>
                  <input
                    type="email"
                    required
                    placeholder="director@facility.com"
                    className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                  />
                </div>

                <div className="pt-3 flex justify-end">
                  <button
                    type="submit"
                    className="w-full py-3 rounded-xl bg-teal-600 hover:bg-teal-500 text-white font-semibold text-sm shadow-md"
                  >
                    Submit Partnership Inquiry
                  </button>
                </div>
              </form>
            </div>
          )}

          {activeModal === 'coordinator' && (
            <div className="space-y-4">
              <div className="flex items-center gap-2 text-teal-600 dark:text-teal-400 text-xs font-semibold uppercase tracking-wider">
                <UserCheck className="w-4 h-4" />
                <span>Care Lead Application</span>
              </div>
              <h3 className="font-heading font-bold text-2xl text-slate-900 dark:text-white">
                Become a RehabConnect Care Coordinator
              </h3>
              <p className="text-sm text-slate-600 dark:text-slate-400">
                Are you a licensed nurse, doctor, or rehabilitation therapist? Join our network of independent patient advocates.
              </p>

              <form
                onSubmit={(e) => {
                  e.preventDefault();
                  alert('Application submitted! Our clinical review board will inspect your credentials within 48 hours.');
                  handleClose();
                }}
                className="space-y-3 pt-2"
              >
                <div>
                  <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                    Your Full Name & Credentials *
                  </label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. Sarah Jenkins, RN, BSN"
                    className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                  />
                </div>

                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                      Clinical Specialty *
                    </label>
                    <input
                      type="text"
                      required
                      placeholder="e.g. Neuro Rehab"
                      className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 dark:text-slate-300 mb-1">
                      Years of Experience *
                    </label>
                    <input
                      type="number"
                      required
                      placeholder="e.g. 8"
                      className="w-full p-3 rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800 text-xs text-slate-900 dark:text-white focus:outline-none focus:border-teal-500"
                    />
                  </div>
                </div>

                <div className="pt-3">
                  <button
                    type="submit"
                    className="w-full py-3 rounded-xl bg-teal-600 hover:bg-teal-500 text-white font-semibold text-sm shadow-md"
                  >
                    Submit Clinical Credentials
                  </button>
                </div>
              </form>
            </div>
          )}
        </motion.div>
      </div>
    </AnimatePresence>
  );
};
