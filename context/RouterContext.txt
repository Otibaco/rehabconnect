'use client'
import React, { createContext, useContext, useEffect, useState } from 'react';
import { RoutePath } from '../lib/types';

interface RouterContextType {
  currentPath: string;
  navigate: (path: string, options?: { scrollToTop?: boolean }) => void;
  activeModal: 'assessment' | 'partner' | 'coordinator' | null;
  openModal: (modal: 'assessment' | 'partner' | 'coordinator') => void;
  closeModal: () => void;
  selectedArticleSlug: string | null;
}

const RouterContext = createContext<RouterContextType | undefined>(undefined);

export const RouterProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [currentPath, setCurrentPath] = useState<string>(() => {
    if (typeof window !== 'undefined') {
      const path = window.location.pathname;
      const hash = window.location.hash.replace('#', '');
      return hash ? `/${hash}` : path || '/';
    }
    return '/';
  });

  const [activeModal, setActiveModal] = useState<'assessment' | 'partner' | 'coordinator' | null>(null);

  // Sync with browser navigation
  useEffect(() => {
    const handlePopState = () => {
      const path = window.location.pathname;
      const hash = window.location.hash.replace('#', '');
      setCurrentPath(hash ? `/${hash}` : path || '/');
    };

    window.addEventListener('popstate', handlePopState);
    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  const navigate = (path: string, options = { scrollToTop: true }) => {
    setCurrentPath(path);
    if (typeof window !== 'undefined') {
      // Use clean pushState & hash for smooth SPA navigation
      window.history.pushState(null, '', path.startsWith('/') ? path : `/${path}`);
      if (options.scrollToTop) {
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }
    }
  };

  const openModal = (modal: 'assessment' | 'partner' | 'coordinator') => {
    setActiveModal(modal);
  };

  const closeModal = () => {
    setActiveModal(null);
  };

  // Derive article slug if on /resources/:slug
  let selectedArticleSlug: string | null = null;
  if (currentPath.startsWith('/resources/') && currentPath !== '/resources') {
    selectedArticleSlug = currentPath.replace('/resources/', '');
  }

  return (
    <RouterContext.Provider
      value={{
        currentPath,
        navigate,
        activeModal,
        openModal,
        closeModal,
        selectedArticleSlug,
      }}
    >
      {children}
    </RouterContext.Provider>
  );
};

export const useRouter = () => {
  const context = useContext(RouterContext);
  if (!context) {
    throw new Error('useRouter must be used within a RouterProvider');
  }
  return context;
};
